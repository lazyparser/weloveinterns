"zh" -}}{{- $fallback = . -}}{{- end -}}{{- end -}} {{- $out := dict -}} {{- range $k := slice "people" "projects" "timeline" "stats" "sources" "positions" -}} {{- $v := index $cur $k -}} {{- if not $v -}}{{- $v = index $fallback $k -}}{{- end -}} {{- if not $v -}}{{- $v = index site.Data $k -}}{{- end -}} {{- $out = merge $out (dict $k $v) -}} {{- end -}} {{- return $out -}}
